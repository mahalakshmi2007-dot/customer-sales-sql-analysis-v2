    order_date DATE,
    customer_name TEXT,
    product TEXT,
    category TEXT,
    region TEXT,
    sales REAL,
    quantity INTEGER,
    profit REAL,
    salesperson TEXT
);

-- 1. View all records
SELECT * FROM sales;

-- 2. Total Revenue
SELECT SUM(sales) AS total_revenue
FROM sales;

-- 3. Total Profit
SELECT SUM(profit) AS total_profit
FROM sales;

-- 4. Highest Sales Order
SELECT order_id, customer_name, product, sales
FROM sales
ORDER BY sales DESC
LIMIT 1;

-- 5. Top 5 Customers
SELECT customer_name, SUM(sales) AS total_spent
FROM sales
GROUP BY customer_name
ORDER BY total_spent DESC
LIMIT 5;

-- 6. Best-Selling Product
SELECT product, SUM(sales) AS total_revenue
FROM sales
GROUP BY product
ORDER BY total_revenue DESC
LIMIT 1;

-- 7. Best Performing Region
SELECT region, SUM(sales) AS total_sales
FROM sales
GROUP BY region
ORDER BY total_sales DESC
LIMIT 1;

-- 8. Category Performance
SELECT category,
       SUM(sales) AS total_sales,
       SUM(profit) AS total_profit
FROM sales
GROUP BY category;

-- 9. Salesperson Performance
SELECT salesperson,
       SUM(sales) AS total_sales,
       SUM(profit) AS total_profit
FROM sales
GROUP BY salesperson;

-- 10. Orders Above Average Sales
SELECT order_id, customer_name, product, sales
FROM sales
WHERE sales > (SELECT AVG(sales) FROM sales)
ORDER BY sales DESC;

-- 11. Sales Classification
SELECT
    CASE
        WHEN sales >= 30000 THEN 'High'
        WHEN sales >= 15000 THEN 'Medium'
        ELSE 'Low'
    END AS sales_category,
    COUNT(*) AS total_orders
FROM sales
GROUP BY sales_category;

-- 12. Monthly Sales
SELECT
    strftime('%Y-%m', order_date) AS month,
    SUM(sales) AS total_sales,
    COUNT(order_id) AS total_orders
FROM sales
GROUP BY month
ORDER BY month;

-- 13. Average Order Value
SELECT AVG(sales) AS average_order_value
FROM sales;

-- 14. Minimum and Maximum Sales
SELECT
    MIN(sales) AS minimum_sales,
    MAX(sales) AS maximum_sales
FROM sales;

-- 15. Sales by Region
SELECT
    region,
    SUM(sales) AS total_sales,
    SUM(profit) AS total_profit,
    COUNT(order_id) AS total_orders
FROM sales
GROUP BY region
ORDER BY total_sales DESC;
